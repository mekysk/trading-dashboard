ซิงค์ข้อมูลข้ามเครื่อง

พอร์ต สมุดเทรด และ watchlist เก็บอยู่ในเบราว์เซอร์ของแต่ละเครื่อง หน้านี้ทำให้มันตามกันไปทุกเครื่อง โดยเข้ารหัสก่อนส่งขึ้นคลาวด์เสมอ ผู้ให้บริการเห็นแต่ก้อนข้อมูลที่อ่านไม่ออก

สถานะตอนนี้

1 · เตรียมที่เก็บ (ทำครั้งเดียว)

1.1 สมัครและสร้างโปรเจกต์ฟรีที่ supabase.com
1.2 เข้าเมนู SQL Editor วางคำสั่งข้างล่างนี้ทั้งก้อน แล้วกด Run
1.3 เข้าเมนู Project Settings → API ก๊อป Project URL กับ anon public key มาใส่ในข้อ 2 

create table if not exists vault (
  id          text primary key,
  data        jsonb not null,
  device      text,
  updated_at  timestamptz not null default now()
);

alter table vault enable row level security;

-- อนุญาตให้คีย์สาธารณะอ่านเขียนได้ แต่เนื้อข้อมูลถูกเข้ารหัสจากเบราว์เซอร์แล้ว
-- ความปลอดภัยจริงอยู่ที่รหัสผ่านของคุณ ไม่ได้อยู่ที่กฎข้อนี้
-- และชื่อแถวเป็นแฮชของรหัสผ่าน คนอื่นจึงหาแถวของคุณไม่เจอ
create policy "vault read"   on vault for select using (true);
create policy "vault write"  on vault for insert with check (true);
create policy "vault update" on vault for update using (true);

2 · ตั้งค่าเครื่องนี้

รหัสผ่านนี้กู้คืนไม่ได้ ไม่มีใครรู้ รวมถึงผมและ Supabase ถ้าลืม ข้อมูลที่เข้ารหัสไว้จะเปิดไม่ได้อีกเลย — จดเก็บไว้ที่ปลอดภัยก่อน
ข้อมูลในเครื่องยังอยู่ครบเสมอ คลาวด์เป็นแค่สำเนา ถ้าลืมรหัสก็แค่ตั้งใหม่แล้วส่งขึ้นไปใหม่

3 · ซิงค์

ปกติไม่ต้องกดเอง ทุกหน้าจะซิงค์ให้ตอนเปิด ปุ่มพวกนี้ไว้ใช้ตอนอยากบังคับทิศทาง

ถ้าทั้งสองฝั่งเปลี่ยนไปคนละทาง ระบบจะไม่เลือกให้ แต่จะบอกให้คุณตัดสินเอง — เลือกผิดแล้วข้อมูลหายกู้ไม่ได้
